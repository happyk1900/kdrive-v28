const modal = document.getElementById('modalChangePass5k'); if (modal) modal.classList.remove('hidden'); } function closeCustomChangePassModal() { const modal = document.getElementById('modalChangePass5k'); const confirmPass = document.getElementById('confirmNewPassVal')?.value.trim(); const btn = document.getElementById('btnSubmitChangePass'); if (!phone) { alert('⚠️ Không tìm thấy số điện thoại của tài khoản!'); return; } if (!newPass || newPass.length < 4) { alert('⚠️ Mật khẩu mới phải có ít nhất 4 ký tự!'); document.getElementById('newPassVal')?.focus(); return; } if (newPass !== confirmPass) { alert('⚠️ Xác nhận mật khẩu mới không khớp!'); if (dispPassEl) dispPassEl.textContent = newPass; localStorage.setItem('kdrive_free_pass', newPass); closeCustomChangePassModal(); showToast("✅ Đã cập nhật mật khẩu trên thiết bị!"); } finally { if (btn) { btn.disabled = false;
